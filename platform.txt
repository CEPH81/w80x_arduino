# W80x IoT Core by MikrocontrollerCephs - plataforma Arduino para WinnerMicro W800, W801, W802,
# W806, Air101 y Air103, sobre WM IoT SDK v2.4 (FreeRTOS).
#
# Especificacion: https://arduino.github.io/arduino-cli/latest/platform-specification/

name=W80x IoT Core by MikrocontrollerCephs
version=0.9.5

# ------------------------------------------------------------------ rutas
sdk.path={runtime.platform.path}/sdk
sdk.chip.path={sdk.path}/{build.soc}

compiler.path={runtime.tools.csky.path}/bin/
compiler.c.cmd=csky-elfabiv2-gcc
compiler.cpp.cmd=csky-elfabiv2-g++
compiler.S.cmd=csky-elfabiv2-gcc
compiler.c.elf.cmd=csky-elfabiv2-g++
compiler.ar.cmd=csky-elfabiv2-ar
compiler.size.cmd=csky-elfabiv2-size

wm_img.path={runtime.platform.path}/tools/wm_img/bin/linux-amd64/wm_img
wm_img.path.windows={runtime.platform.path}/tools/wm_img/bin/windows-amd64/wm_img.exe
wm_img.path.macosx={runtime.platform.path}/tools/wm_img/bin/darwin-amd64/wm_img

# ------------------------------------------------------------------ opciones
compiler.warning_flags=-w
compiler.warning_flags.none=-w
compiler.warning_flags.default=
compiler.warning_flags.more=-Wall
compiler.warning_flags.all=-Wall -Wextra

build.flags.optimize=-O2
build.extra_flags=
build.log_level=2

compiler.cpu_flags=-mcpu=ck804ef -mhard-float
compiler.common_flags={compiler.cpu_flags} {build.flags.optimize} -g -ffunction-sections -fdata-sections -fno-builtin-printf -fno-builtin-fprintf
compiler.defines=-DF_CPU={build.f_cpu} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} -DW80X_CHIP_{build.soc} -DW80X_LOG_LEVEL={build.log_level} "-DMBEDTLS_CONFIG_FILE=\"wm_mbedtls_config.h\"" {build.extra_flags}
compiler.sdk.includes="-iprefix{sdk.path}/include/" "@{sdk.chip.path}/includes.txt" "-I{sdk.chip.path}/config" "-I{sdk.chip.path}/device_table"

compiler.c.flags=-c {compiler.common_flags} -std=gnu99 {compiler.warning_flags} -MMD
compiler.cpp.flags=-c {compiler.common_flags} -std=gnu++14 -fno-rtti -fno-exceptions -fno-threadsafe-statics {compiler.warning_flags} -MMD
compiler.S.flags=-c {compiler.cpu_flags} -x assembler-with-cpp -MMD
compiler.ar.flags=rcs
compiler.c.elf.flags={compiler.cpu_flags} -nostartfiles -Wl,--gc-sections -Wl,-zmax-page-size=1024

compiler.c.extra_flags=
compiler.cpp.extra_flags=
compiler.S.extra_flags=
compiler.ar.extra_flags=
compiler.c.elf.extra_flags=
compiler.libraries.ldflags=

# ------------------------------------------------------------------ compilacion
recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} {compiler.defines} {compiler.c.extra_flags} {includes} {compiler.sdk.includes} "{source_file}" -o "{object_file}"
recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} {compiler.defines} {compiler.cpp.extra_flags} {includes} {compiler.sdk.includes} "{source_file}" -o "{object_file}"
recipe.S.o.pattern="{compiler.path}{compiler.S.cmd}" {compiler.S.flags} {compiler.defines} {compiler.S.extra_flags} {includes} {compiler.sdk.includes} "{source_file}" -o "{object_file}"

archive_file_path={build.path}/{archive_file}
recipe.ar.pattern="{compiler.path}{compiler.ar.cmd}" {compiler.ar.flags} {compiler.ar.extra_flags} "{archive_file_path}" "{object_file}"

# enlace: el nucleo del SDK completo (arranque, interrupciones, drivers, RTOS)
# y el resto (WiFi, lwIP, mbedTLS...) solo si el programa lo usa
recipe.c.combine.pattern="{compiler.path}{compiler.c.elf.cmd}" {compiler.c.elf.flags} {compiler.c.elf.extra_flags} "-T{sdk.chip.path}/ld/chip_layout.ld" "-Wl,-Map={build.path}/{build.project_name}.map" -o "{build.path}/{build.project_name}.elf" {object_files} -Wl,--start-group "{archive_file_path}" {compiler.libraries.ldflags} "-L{sdk.chip.path}/lib" -Wl,--whole-archive "@{sdk.chip.path}/libs_whole.txt" -Wl,--no-whole-archive "@{sdk.chip.path}/libs.txt" -lstdc++ -lsupc++ -lm -lc -lgcc -Wl,--end-group

# imagen (.img) y archivo de carga (.fls: bootloader + programa + tabla de particiones)
recipe.objcopy.fls.pattern="{wm_img.path}" -sdk "{sdk.chip.path}" -elf "{build.path}/{build.project_name}.elf" -img "{build.path}/{build.project_name}.img" -fls "{build.path}/{build.project_name}.fls" -ck "{build.path}/{build.project_name}.ck.bin"

recipe.output.tmp_file={build.project_name}.fls
recipe.output.save_file={build.project_name}.{build.variant}.fls

recipe.size.pattern="{compiler.path}{compiler.size.cmd}" -A "{build.path}/{build.project_name}.elf"
recipe.size.regex=^(?:\.text|\.rodata|\.data)\s+([0-9]+).*
recipe.size.regex.data=^(?:\.vdata|\.data|\.bss)\s+([0-9]+).*

# ------------------------------------------------------------------ carga por puerto serie (wm_tool)
# wm_img lanza wm_tool y, al terminar, saca al chip de la ROM de carga con la orden de
# reinicio del SDK (wm_tool lo deja en modo de carga en las placas HLK y habia que pulsar RST)
tools.serial_upload.wm_tool.windows={runtime.tools.w80x_tool.path}/win/wm_tool.exe
tools.serial_upload.wm_tool.linux={runtime.tools.w80x_tool.path}/lin/wm_toold
tools.serial_upload.cmd={wm_img.path}
tools.serial_upload.upload.params.verbose=
tools.serial_upload.upload.params.quiet=
tools.serial_upload.upload.pattern="{cmd}" serial -tool "{wm_tool}" -port {serial.port} -- -c {serial.port} -rs rts -ws 115200 -ds {upload.speed} {upload.erase} -dl "{build.path}/{build.project_name}.fls"

# ------------------------------------------------------------------ carga por CK-Link (XuanTie Debugger Server)
tools.cklink.cmd={runtime.platform.path}/tools/wm_img/bin/linux-amd64/wm_img
tools.cklink.cmd.windows={runtime.platform.path}/tools/wm_img/bin/windows-amd64/wm_img.exe
tools.cklink.cmd.macosx={runtime.platform.path}/tools/wm_img/bin/darwin-amd64/wm_img
tools.cklink.upload.params.verbose=
tools.cklink.upload.params.quiet=
tools.cklink.upload.pattern="{cmd}" cklink -chip {build.soc} -file "{build.path}/{build.project_name}.ck.bin"
tools.cklink.program.params.verbose=
tools.cklink.program.params.quiet=
tools.cklink.program.pattern="{cmd}" cklink -chip {build.soc} -file "{build.path}/{build.project_name}.ck.bin"
